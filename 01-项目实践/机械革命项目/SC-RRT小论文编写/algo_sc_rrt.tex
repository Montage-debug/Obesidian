% ============================================================
%  Algorithm 1  ADCS -- Adaptive Dual-Ellipsoid Constrained Sampling
%  Based on: ADCS.png flowchart
% ============================================================
\begin{algorithm}
\caption{ADCS: Adaptive Dual-Ellipsoid Constrained Sampling}
\begin{algorithmic}[1]
\Require tree focus $x_f$, mediator $x_m$, ellipsoid bound $c$,
         expansion factor $\gamma$, informed ratio $p$,
         cross-tree ratio $\rho$, opposite tree $T_{opp}$
\Ensure  sample $x_{rand}$

\State Draw $r \sim \text{Uniform}[0, 1]$

\If{$r < p$ and $c < \infty$}
  \State \textbf{[Informed branch]}
  \State Compute ellipsoid $E$ with foci $x_f$, $x_m$ and semi-major axis $\gamma \cdot c$
  \State $x_{rand} \leftarrow$ uniform sample inside $E$
  \Comment{constrained to scaled dual-ellipsoid}

\ElsIf{$r < p + (1 - p) \cdot \rho$}
  \State \textbf{[Cross-tree branch]}
  \State Select target node $x_t$ from Pareto front of $T_{opp}$
  \State $x_{rand} \leftarrow x_t +$ small Gaussian noise
  \Comment{biased toward high-quality nodes in opposite tree}

\Else
  \State \textbf{[Global branch]}
  \State $x_{rand} \leftarrow$ uniform sample over workspace bounds
  \Comment{maintain global exploration coverage}
\EndIf

\Return $x_{rand}$
\end{algorithmic}
\end{algorithm}


% ============================================================
%  Algorithm 2  SSFOR -- Search-State Feedback Online Regulation
%  Based on: SSFOR.png flowchart
% ============================================================
\begin{algorithm}
\caption{SSFOR: Search-State Feedback Online Regulation}
\begin{algorithmic}[1]
\Require current best cost $L^*$, cost history $H$, controller state $s$,
         window size $W$, target improvement rate $y^*$
\Ensure  updated expansion factor $\gamma$, informed ratio $p$, state $s$

\If{$L^* = \infty$ or $|H| \leq W$}
  \State No feasible solution yet
  \Return $\gamma \leftarrow \gamma_{max}$, $p \leftarrow 0$
  \Comment{full exploration: maximize ellipsoid, disable informed sampling}
\EndIf

\State \textbf{[Step 1: Measure recent improvement]}
\State $y \leftarrow (H[-W] - L^*) \;/\; H[-W]$
\Comment{sliding-window path cost improvement rate}
\State $\bar{y} \leftarrow (1 - \alpha) \cdot \bar{y}_{prev} + \alpha \cdot y$
\Comment{exponential moving average smoothing}

\State \textbf{[Step 2: Compute control signal via PID]}
\State $e \leftarrow y^* - \bar{y}$
\Comment{error: gap between target and actual improvement rate}
\State $u \leftarrow K_p \cdot e \;+\; K_i \cdot s.I \;+\; K_d \cdot (e - s.e_{prev})$
\State $s.I \leftarrow \text{clip}(s.I + e,\; [I_{min},\; I_{max}])$
\Comment{anti-windup integral update}

\State \textbf{[Step 3: Regulate sampling parameters (inverse coupling)]}
\If{$e > 0$}
  \State Increase $\gamma$ (expand ellipsoid), decrease $p$ (more global sampling)
  \Comment{improvement stagnant: widen search space}
\Else
  \State Decrease $\gamma$ (shrink ellipsoid), increase $p$ (more informed sampling)
  \Comment{improvement active: focus on promising region}
\EndIf

\State Clip $\gamma$ to $[\gamma_{min},\; \gamma_{max}]$,\; clip $p$ to $[p_{min},\; p_{max}]$
\State Update $s.e_{prev} \leftarrow e$

\Return $\gamma$, $p$, $s$
\end{algorithmic}
\end{algorithm}


% ============================================================
%  Algorithm 3  SC-RRT Main Loop
% ============================================================
\begin{algorithm}
\caption{SC-RRT: Bidirectional Path Planning}
\begin{algorithmic}[1]
\Require start $x_s$, goal $x_g$, max iterations $N$, update interval $T_u$
\Ensure  planned path, or Failure

\State Initialize trees $T_A$ from $x_s$,\; $T_B$ from $x_g$
\State $x_m \leftarrow$ midpoint$(x_s,\, x_g)$;\; $L^* \leftarrow \infty$;\; $H \leftarrow []$;\; $s \leftarrow$ init

\For{$k = 1$ to $N$}
  \If{$k \bmod T_u = 0$}
    \State $x_m \leftarrow$ UpdateMeetPoint$(T_A,\, T_B)$
    \Comment{online mediator update via weighted centroid}
    \State $(c_A,\, c_B) \leftarrow$ DualEllipsoidParams$(T_A,\, T_B,\, x_m)$
    \Comment{independent bound for each tree side}
    \State $(\gamma,\, p,\, s) \leftarrow$ SSFOR$(L^*,\, H,\, s)$
    \Comment{regulate $\gamma$ and $p$ based on search progress}
  \EndIf

  \For{each $(T,\; x_f,\; c)$ in $\{(T_A,\, x_s,\, c_A),\; (T_B,\, x_g,\, c_B)\}$}
    \State $x_{rand} \leftarrow$ ADCS$(x_f,\, x_m,\, c,\, \gamma,\, p,\, T_{opp})$
    \Comment{three-way informed sampling}
    \State $x_{near} \leftarrow$ Nearest$(T,\, x_{rand})$;\;\;
           $x_{new} \leftarrow$ Steer$(x_{near},\, x_{rand})$
    \If{CollisionFree$(x_{near},\, x_{new})$}
      \State Add $x_{new}$ to $T$ with best parent in local neighborhood
      \Comment{RRT* rewiring}
      \If{TryConnect$(x_{new},\, T_{opp})$ succeeds}
        \State $L^* \leftarrow$ cost of extracted path;\;\; append $L^*$ to $H$
      \EndIf
    \EndIf
  \EndFor
\EndFor

\If{$L^* < \infty$}
  \Return SmoothPath(ExtractPath$(T_A,\, T_B)$)
\Else
  \Return Failure
\EndIf
\end{algorithmic}
\end{algorithm}

\begin{algorithm}
\caption{SC-RRT: Bidirectional Path Planning}
\begin{algorithmic}[1]
\Require start $x_s$, goal $x_g$, max iterations $N$, update interval $T_u$
\Ensure planned path or Failure
\State Initialize trees $T_A$ from $x_s$, $T_B$ from $x_g$
\State $x_m \leftarrow$ midpoint$(x_s, x_g)$; best cost $L^* \leftarrow \infty$; SSFOR state $s \leftarrow$ init
\For{$k = 1$ to $N$}
  \If{$k \bmod T_u = 0$}
    \State $x_m \leftarrow$ UpdateMeetPoint$(T_A, T_B)$ \Comment{potential mediator via weighted centroid}
    \State $(c_A, c_B) \leftarrow$ DualEllipsoidParams$(T_A, T_B, x_m)$ \Comment{non-symmetric ellipsoid bounds}
    \State $(\gamma, p, s) \leftarrow$ SSFOR$(L^*, \mathcal{H}, s)$ \Comment{regulate expansion factor and informed ratio}
  \EndIf
  \For{each tree $(T, x_f, c)$ in $\{(T_A, x_s, c_A),\ (T_B, x_g, c_B)\}$}
    \State $x_{rand} \leftarrow$ AdaptiveSample$(x_f, x_m, c, \gamma, p, T_{opp})$ \Comment{ADCS three-way sampling}
    \State $x_{near} \leftarrow$ Nearest$(T, x_{rand})$;\ \ $x_{new} \leftarrow$ Steer$(x_{near}, x_{rand})$
    \If{CollisionFree$(x_{near}, x_{new})$}
      \State Add $x_{new}$ to $T$ with best parent in neighborhood \Comment{RRT* rewiring}
      \If{TryConnect$(x_{new}, T_{opp})$ succeeds}
        \State Update $L^*$ and append to history $\mathcal{H}$
      \EndIf
    \EndIf
  \EndFor
\EndFor
\If{$L^* < \infty$}\ \Return SmoothPath(ExtractPath$(T_A, T_B)$)
\Else\ \Return Failure
\EndIf
\end{algorithmic}
\end{algorithm}


% ============================================================
%  Algorithm 2  ADCS -- Adaptive Sampling
% ============================================================
\begin{algorithm}
\caption{AdaptiveSample (ADCS): Three-Way Informed Sampling}
\begin{algorithmic}[1]
\Require focus $x_f$, mediator $x_m$, ellipsoid bound $c$, expansion $\gamma$,
         informed ratio $p$, cross-tree ratio $\rho$, opposite tree $T_{opp}$
\Ensure sample point $x_{rand}$
\State Draw $r \sim$ Uniform$[0, 1]$
\If{$r < p$ and $c < \infty$}
  \State $x_{rand} \leftarrow$ SampleInEllipsoid$(x_f,\ x_m,\ \gamma \cdot c)$
  \Comment{informed: constrained to scaled ellipsoid}
\ElsIf{$r < p + (1-p) \cdot \rho$}
  \State $x_{rand} \leftarrow$ ParetoNode$(T_{opp})$ + small noise
  \Comment{cross-tree: biased toward opposite tree's Pareto front}
\Else
  \State $x_{rand} \leftarrow$ UniformSample(workspace bounds)
  \Comment{global: uniform exploration}
\EndIf
\Return $x_{rand}$
\end{algorithmic}
\end{algorithm}


% ============================================================
%  Algorithm 3  SSFOR -- Online Regulation
% ============================================================
\begin{algorithm}
\caption{SSFOR: Search-State Feedback Online Regulation}
\begin{algorithmic}[1]
\Require current best cost $L^*$, cost history $\mathcal{H}$, controller state $s$
\Ensure updated expansion factor $\gamma$, informed ratio $p$, state $s$
\If{no feasible solution yet or history too short}
  \Return max expansion $\gamma$, zero informed ratio $p$ \Comment{full exploration phase}
\EndIf
\State Compute recent improvement rate $y$ from sliding window over $\mathcal{H}$
\State Smooth $y$ with exponential moving average
\State Compute control error $e \leftarrow$ target rate $-$ smoothed rate $y$
\State Compute PID output $u$ from error $e$ and state $s$
\If{$e > 0$} \Comment{improvement stagnant: expand search}
  \State Increase $\gamma$ (enlarge ellipsoid); decrease $p$ (more global sampling)
\Else \Comment{improvement active: tighten constraint}
  \State Decrease $\gamma$ (shrink ellipsoid); increase $p$ (more informed sampling)
\EndIf
\State Update state $s$; clip $\gamma$ and $p$ to valid ranges
\Return $\gamma$, $p$, $s$
\end{algorithmic}
\end{algorithm}
