#!/bin/bash
# generate-global-asoundconf.sh

set -e

echo "🔍 检测 USB 音频设备..."

if [ ! -f /proc/asound/cards ]; then
    echo "❌ ALSA 未安装或无音频子系统。"

CARD_NAME=$(echo "$USB_CARD_LINE" | sed -n 's/.*\[\(.*\)\].*/\1/p' | xargs)
[ -z "$CARD_NAME" ] && CARD_NAME="Device"

# =============================================
# 全局 ALSA 配置：USB 音频多用户/多进程支持
# 设备: $CARD_NAME (Card $CARD_ID)
# 生成时间: $(date)
# =============================================

# 播放混音器
pcm.dmix_usb {
    type dmix
    ipc_key 1024
    ipc_perm 0666
    slave {
        pcm "hw:$CARD_ID"
        rate 48000
        period_size 1024
        buffer_size 4096
        periods 4
    }
}

# 录音共享器
pcm.dsnoop_usb {
    type dsnoop
    ipc_key 1025
    ipc_perm 0666
    slave {
        pcm "hw:$CARD_ID"
        rate 16000
        channels 1
        period_size 1024
        buffer_size 4096
    }
}

# 合并播放与录音
pcm.usb {
    type asym
    playback.pcm "dmix_usb"
    capture.pcm "dsnoop_usb"
}

# 默认设备
pcm.!default {
    type plug
    slave.pcm "usb"
}

ctl.!default {
    type hw
    card $CARD_ID
}
ASOUNDEOF

#!/usr/bin/env python3
import sys
import subprocess
import os
import json
import importlib
import importlib.metadata
from typing import Tuple, List, Dict

def check_package(package_name: str, min_version: str = None) -> Tuple[bool, str]:
    """检查单个包的安装情况"""
    try:
        # 包名到模块名的映射
        package_to_module = {
            "python-jose": "jose",
            "pysqlcipher3-binary": "pysqlcipher3",
        }
        
        # 特殊包处理
        if package_name == "elite-cs-series-sdk":
            if os.path.exists("/usr/include/Elite/") or os.path.exists("/opt/elite-sdk/"):
                return True, "SDK installed"
            return False, "SDK not found"
        
        elif "ros-humble-" in package_name:
            try:
                result = subprocess.run(
                    ["ros2", "pkg", "list"],
                    capture_output=True,
                    text=True,
                    timeout=5
                )
                if package_name.replace("ros-humble-", "") in result.stdout:
                    return True, "ROS package installed"
            except:
                pass
            return False, "ROS package not found"
        
        # 获取实际的模块名
        module_name = package_to_module.get(package_name, package_name)
        
        # 使用importlib.metadata检查包版本（Python 3.8+）
        try:
            # 首先尝试使用importlib.metadata
            version = importlib.metadata.version(package_name)
        except importlib.metadata.PackageNotFoundError:
            # 如果失败，尝试直接导入模块
            try:
                # 抑制导入时的输出
                import io
                import contextlib
                
                f = io.StringIO()
                with contextlib.redirect_stdout(f), contextlib.redirect_stderr(f):
                    if '.' in module_name:
                        parts = module_name.split('.')
                        mod = __import__(parts[0])
                        for part in parts[1:]:
                            mod = getattr(mod, part, None)
                            if mod is None:
                                return False, "Submodule not found"
                        version = getattr(mod, '__version__', 'installed')
                    else:
                        mod = __import__(module_name)
                        version = getattr(mod, '__version__', 'installed')
            except ImportError:
                return False, "Not installed"
            except Exception:
                return False, "Check failed"
        
        # 版本检查
        if min_version and version != 'installed':
            try:
                from packaging import version as pkg_version
                if pkg_version.parse(str(version)) >= pkg_version.parse(min_version):
                    return True, f"{version} ✓"
                else:
                    return True, f"{version} (needs {min_version}+)"
            except:
                return True, str(version)
        
        return True, str(version)
        
    except Exception as e:
        return False, f"Check error"

# 其余的函数保持不变（check_system_package, check_audio_config等）
def check_system_package(package_name: str) -> Tuple[bool, str]:
    """检查系统包是否安装"""
    try:
        result = subprocess.run(
            ["dpkg", "-l", package_name],
            capture_output=True,
            text=True
        )
        if "ii" in result.stdout:
            lines = result.stdout.strip().split('\n')
            for line in lines:
                if line.startswith("ii"):
                    parts = line.split()
                    if len(parts) >= 3:
                        return True, parts[2]
            return True, "installed"
        return False, "not installed"
    except Exception as e:
        return False, f"Check error: {str(e)}"

def check_audio_config() -> Tuple[bool, str]:
    """检查音频配置"""
    checks = []
    
    alsa_packages = ["alsa-base", "alsa-utils", "libasound2"]
    for pkg in alsa_packages:
        success, msg = check_system_package(pkg)
        checks.append(f"ALSA {pkg}: {'✅' if success else '❌'} {msg}")
    
    script_path = "/opt/scripts/generate-global-asoundconf.sh"
    if os.path.exists(script_path):
        checks.append(f"Audio script: ✅ {script_path}")
    else:
        checks.append(f"Audio script: ❌ Not found")
    
    try:
        result = subprocess.run(
            ["systemctl", "is-enabled", "rc-local.service"],
            capture_output=True,
            text=True
        )
        if result.returncode == 0:
            checks.append(f"rc-local service: ✅ {result.stdout.strip()}")
        else:
            checks.append("rc-local service: ❌ Not enabled")
    except:
        checks.append("rc-local service: ❌ Check failed")
    
    # 检查/etc/asound.conf
    if os.path.exists("/etc/asound.conf"):
