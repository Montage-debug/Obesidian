# 二、脚本文件内容
```c
# monitor_egress_domains.sh 脚本内容
#!/bin/bash

# ==============================================================================
# 脚本名称: monitor_egress_domains.sh
# 功能: 监听网络出口域名 (通过 DNS 和 TLS SNI)，统计并输出到日志文件。
# 适用系统: Ubuntu 22.04+
# 用途: 物联网卡白名单统计
# ==============================================================================

# 配置区域
# 请修改为你的主要网卡名称 (如 eth0, ens33, wlan0 等)。
# 如果留空，脚本会自动尝试检测默认路由网卡。
TARGET_INTERFACE="" 

# 输出日志文件路径
LOG_FILE="/var/log/egress_domains.log"
# 去重后的白名单文件路径
WHITELIST_FILE="./iot_whitelist.txt"

# 检查是否以 root 运行
if [ "$EUID" -ne 0 ]; then 
  echo "错误：请使用 sudo 运行此脚本 (sudo ./monitor_egress_domains.sh)"
  exit 1
fi

# 自动检测网卡 (如果未指定)
if [ -z "$TARGET_INTERFACE" ]; then
    TARGET_INTERFACE=$(ip route | grep default | awk '{print $5}' | head -n 1)
    if [ -z "$TARGET_INTERFACE" ]; then
        echo "错误：无法自动检测到活动网卡，请手动在脚本中设置 TARGET_INTERFACE 变量。"
        exit 1
    fi
    echo "未指定网卡，已自动检测到默认网卡: $TARGET_INTERFACE"
fi

echo "开始监听网卡: $TARGET_INTERFACE"
echo "原始日志将保存至: $LOG_FILE"
echo "按 Ctrl+C 停止监听并生成白名单..."
echo "------------------------------------------------------------------------------"

# 确保日志文件存在
touch "$LOG_FILE"

# 定义清理函数，用于在退出时生成白名单
cleanup() {
    echo ""
    echo "正在停止监听..."
    echo "正在从日志中提取唯一域名并生成白名单: $WHITELIST_FILE"
    
    # 处理逻辑：
    # 1. 读取日志
    # 2. 提取域名部分 (去除时间戳和前缀)
    # 3. 排序并去重
    # 4. 过滤掉明显的本地域名或无效数据
    cat "$LOG_FILE" | \
    awk '{print $NF}' | \
    grep -E '^[a-zA-Z0-9.-]+\.[a-zA-Z]{2,}$' | \
    grep -v '\.local$' | \
    sort -u > "$WHITELIST_FILE"
    
    COUNT=$(wc -l < "$WHITELIST_FILE")
    echo "完成！共发现 $COUNT 个唯一域名。"
    echo "白名单文件已保存至当前目录下的: $WHITELIST_FILE"
    exit 0
}

trap cleanup SIGINT SIGTERM

# 启动 tshark 进行监听
# 过滤条件:
#   dns.flags.response == 1  : 捕获 DNS 响应包 (获取查询的域名)
#   tls.handshake.type == 1  : 捕获 TLS 客户端握手包 (获取 SNI 域名，即使没有DNS缓存也能抓到)
# 输出格式: 只输出域名字段
sudo tshark -i "$TARGET_INTERFACE" \
    -Y "dns.flags.response == 1 or tls.handshake.type == 1" \
    -T fields \
    -e dns.qry.name \
    -e tls.handshake.extensions_server_name \
    -E separator=, \
    2>/dev/null | while read -r line; do
    
    # 解析 tshark 的输出 (格式可能是 "domain.com," 或 ",domain.com" 或 "domain1,domain2")
    # 我们只需要提取出有效的域名部分
    
    # 分割逗号分隔的字段
    IFS=',' read -ra DOMAINS <<< "$line"
    
    for domain in "${DOMAINS[@]}"; do
        # 去除空白字符
        domain=$(echo "$domain" | xargs)
        
        # 简单的有效性验证：必须包含点，且不为空
        if [[ -n "$domain" && "$domain" == *"."* ]]; then
            # 记录时间戳和域名
            TIMESTAMP=$(date +"%Y-%m-%d %H:%M:%S")
            echo "[$TIMESTAMP] $domain" | tee -a "$LOG_FILE"
        fi
    done
done
```
