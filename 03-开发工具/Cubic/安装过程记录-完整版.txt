#!/bin/bash
# generate-global-asoundconf.sh

set -e

echo "🔍 检测 USB 音频设备..."

if [ ! -f /proc/asound/cards ]; then
    echo "❌ ALSA 未安装或无音频子系统。"
    exit 1
fi

USB_CARD_LINE=$(grep -i usb /proc/asound/cards | head -n1)
if [ -z "$USB_CARD_LINE" ]; then
    echo "❌ 未找到 USB 音频设备。请插入后重试。"
    cat /proc/asound/cards
    exit 1
fi

CARD_ID=$(echo "$USB_CARD_LINE" | awk '{print $1}')
CARD_NAME=$(echo "$USB_CARD_LINE" | sed -n 's/.*\[\(.*\)\].*/\1/p' | xargs)
[ -z "$CARD_NAME" ] && CARD_NAME="Device"

echo "✅ 使用设备: $CARD_NAME (Card ID: $CARD_ID)"
echo "ℹ️  实际硬件路径: hw:$CARD_ID"

# 生成 /etc/asound.conf
cat > /etc/asound.conf << ASOUNDEOF
# =============================================
# 全局 ALSA 配置：USB 音频多用户/多进程支持
# 设备: $CARD_NAME (Card $CARD_ID)
# 生成时间: $(date)
# =============================================

# 播放混音器
pcm.dmix_usb {
    type dmix
    ipc_key 1024
    ipc_perm 0666
    slave {
        pcm "hw:$CARD_ID"
        rate 48000
        period_size 1024
        buffer_size 4096
        periods 4
    }
}

# 录音共享器
pcm.dsnoop_usb {
    type dsnoop
    ipc_key 1025
    ipc_perm 0666
    slave {
        pcm "hw:$CARD_ID"
        rate 16000
        channels 1
        period_size 1024
        buffer_size 4096
    }
}

# 合并播放与录音
pcm.usb {
    type asym
    playback.pcm "dmix_usb"
    capture.pcm "dsnoop_usb"
}

# 默认设备
pcm.!default {
    type plug
    slave.pcm "usb"
}

ctl.!default {
    type hw
    card $CARD_ID
}
ASOUNDEOF

echo "✅ 全局 ALSA 配置已写入 /etc/asound.conf"

    echo "✅ /dev/snd 权限已更新"
fi

alsactl restore 2>/dev/null || true
echo "ℹ️  确保用户在 'audio' 组"
EOF

#!/usr/bin/env python3
import sys
import subprocess

def check_package(package_name, min_version=None):
    """检查单个包的安装情况"""
    try:
        if '.' in package_name:
            # 处理有子模块的包，如pyarmor.cli.runtime
            parts = package_name.split('.')
            mod = __import__(parts[0])
            for part in parts[1:]:
                mod = getattr(mod, part, None)
                if mod is None:
                    return False, "未找到子模块"
            version = getattr(mod, '__version__', '已安装')
            return True, version
        else:
            mod = __import__(package_name)
            version = getattr(mod, '__version__', '已安装')
            
            if min_version and version != '已安装':
                # 简单的版本比较
                try:
                    from packaging import version as pkg_version
                    if pkg_version.parse(str(version)) >= pkg_version.parse(min_version):
                        return True, f"{version} ✓"
                    else:
                        return True, f"{version} (需要 {min_version}+)"
                except:
                    return True, version
            return True, version
    except ImportError as e:
        return False, f"未安装 ({str(e)})"
    except Exception as e:
        return False, f"检查失败 ({str(e)})"

def main():
    print("=" * 60)
    print("全局环境验证")
    print("=" * 60)
    
    # 检查Python和pip版本
    python_version = sys.version.split()[0]
    print(f"Python版本: {python_version}")
    
    try:
        import pip
        print(f"Pip版本: {pip.__version__}")
    except:
        print("Pip版本: 未知")
    
    print("\n" + "-" * 60)
    print("包安装状态:")
    print("-" * 60)
    
    # 要检查的包列表（从requirements.txt提取的关键包）
    packages_to_check = [
        ("fastapi", "0.104.1"),
        ("uvicorn", "0.24.0"),
        ("pydantic", "2.5.0"),
        ("sqlalchemy", "2.0.23"),
        ("openai", "2.14.0"),
        ("numpy", "1.20.0"),
        ("pygame", "2.6.1"),
        ("pydub", "0.25.1"),
        ("websockets", "12.0"),
        ("httpx", "0.25.1"),
        ("aiosqlite", None),
        ("python-jose", None),  # python-jose导入名
        ("pyarmor", None),      # pyarmor.cli.runtime的父包
        ("lark", None),
        ("empy", None),
    ]
    
    results = []
    all_ok = True
    
    for package, min_version in packages_to_check:
        success, message = check_package(package, min_version)
        
        if success:
            status = "✅"
            results.append(f"{status} {package:25} {message}")
        else:
            status = "❌"
            results.append(f"{status} {package:25} {message}")
            all_ok = False
        
        # 显示进度
        sys.stdout.write(f"\r检查: {package:25} {'✓' if success else '✗'}")
        sys.stdout.flush()
    
    print("\n" + "-" * 60)
    
    # 显示结果
    for result in results:
        print(result)
    
    print("-" * 60)
    
    # 检查包安装位置
    print("\n包安装位置:")
    try:
        import site
        for path in site.getsitepackages():
            print(f"  {path}")
    except:
        pass
    
    # 最终结论
    if all_ok:
        print("\n🎉 所有包安装成功！环境就绪。")
        return 0
    else:
        print("\n⚠ 部分包安装有问题，但核心功能应该可用。")
        return 1

if __name__ == "__main__":
    sys.exit(main())
EOF

# Ubuntu自动安装配置 - Gxy
# =============================================

# 基础设置
d-i debian-installer/locale string en_US
d-i keyboard-configuration/xkb-keymap select us
d-i netcfg/choose_interface select auto

# 分区设置
d-i partman-auto/method string regular
d-i partman-auto/choose_recipe select atomic
d-i partman/confirm boolean true
d-i partman/confirm_nooverwrite boolean true

# 用户设置
d-i passwd/user-fullname string Ubuntu User
d-i passwd/username string ubuntu
d-i passwd/user-password password ubuntu
d-i passwd/user-password-again password ubuntu
d-i user-setup/allow-password-weak boolean true

# 软件源和包管理
d-i apt-setup/extras boolean true
# 添加更多基础包
d-i pkgsel/include string \
    openssh-server \
    build-essential \
    git \
    curl \
    wget \
    python3 \
    python3-pip \
    python3-dev \
    python3-venv \
    ffmpeg \
    libssl-dev \
    libffi-dev \
    pkg-config \
    alsa-base \
    alsa-utils \
    libasound2-dev \
    libusb-1.0-0-dev \
    libsqlcipher-dev \
    sqlcipher \
    software-properties-common

# 软件包选择
tasksel tasksel/first multiselect ubuntu-desktop

# 保留安装的包
ubiquity ubiquity/keep-installed string icedtea6-plugin openoffice.org

# =============================================
# 安装后自动执行自定义配置
# =============================================
d-i preseed/late_command string \
    # 1. 复制custom配置到新系统 \
    cp -r /cdrom/custom /target/opt/; \
    chmod -R a+rX /target/opt/custom; \
    \
    # 2. 运行安装后配置脚本 \
    in-target /opt/custom/install-post.sh; \
    \
    # 3. 设置Python包权限 \
    in-target find /usr/local/lib/python3.* -name "dist-packages" -type d -exec chmod -R a+rX {} \;; \
    in-target chmod -R a+rX /usr/local/bin; \
    \
    # 4. 启用系统服务 \
    in-target systemctl enable ssh; \
    in-target systemctl enable rc-local.service;

# 完成安装
d-i finish-install/reboot_in_progress note
#==================U盘配置====================
# 查看U盘设备
sudo fdisk -l
# 卸载U盘
sudo umount /dev/sdX*

# 烧录ISO
sudo dd if=fastapi-ubuntu.iso of=/dev/sdc bs=4M status=progress oflag=sync

# 同步
sudo sync


#  使用Ventoy
#  安装到U盘
#  将ISO文件复制到U盘
#  启动时选择ISO文件
#  开机进入BIOS/UEFI

#==================验证=======================
# 直接运行验证脚本
verify-fastapi
